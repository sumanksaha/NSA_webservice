  27| def _load_or_create_production_secret_key(app: Flask) -> str:
  28|     """Return a stable production SECRET_KEY, generating + persisting one if needed.
  29| 
  30|     Render's ``generateValue: true`` only mints a value when the env var is
  31|     FIRST created on the service, so services that predate the setting (or
  32|     were created from the dashboard) can boot with no SECRET_KEY. Instead of
  33|     crashing the deploy, generate a strong key once and persist it so
  34|     sessions stay valid across restarts and redeploys.
  35| 
  36|     Persistence order (first success wins):
  37|       1. ``app_secrets`` key/value table in the primary DB (survives
  38|          redeploys on Render's persistent Postgres).
  39|       2. ``<instance_path>/.secret_key`` file (survives restarts).
  40|       3. Ephemeral in-memory key — sessions reset on restart, but the app
  41|          still boots (never blocks a deploy).
  42| 
  43|     An explicit ``SECRET_KEY`` env var (dashboard-managed or a
  44|     generateValue-minted value) always takes precedence and bypasses this.
  45|     """
  46|     database_url = os.environ.get("DATABASE_URL")
  47|     if database_url:
  48|         # Normalize postgres:// -> postgresql:// (same as create_app does)
  49|         if database_url.startswith("postgres://"):
  50|             database_url = database_url.replace("postgres://", "postgresql://", 1)
  51|         try:
  52|             from sqlalchemy import create_engine, text
  53| 
  54|             engine = create_engine(database_url)
  55|             with engine.begin() as conn:
  56|                 conn.execute(
  57|                     text(
  58|                         "CREATE TABLE IF NOT EXISTS app_secrets (name VARCHAR(64) PRIMARY KEY, value TEXT NOT NULL)",
  59|                     ),
  60|                 )
  61|                 row = conn.execute(
  62|                     text("SELECT value FROM app_secrets WHERE name = 'secret_key'"),
  63|                 ).fetchone()
  64|                 if row:
  65|                     app.logger.info("SECRET_KEY auto-provisioned (DB — reuse)")
  66|                     return str(row[0])
  67|                 new_key = secrets.token_hex(32)
  68|                 conn.execute(
  69|                     text(
  70|                         "INSERT INTO app_secrets (name, value) VALUES ('secret_key', :v) ON CONFLICT(name) DO NOTHING",
  71|                     ),
  72|                     {"v": new_key},
  73|                 )
  74|                 row = conn.execute(
  75|                     text("SELECT value FROM app_secrets WHERE name = 'secret_key'"),
  76|                 ).fetchone()
  77|                 app.logger.info("SECRET_KEY auto-provisioned (DB — new)")
  78|                 return str(row[0]) if row else new_key
  79|         except Exception as exc:
  80|             app.logger.warning("SECRET_KEY DB persistence unavailable: %s", exc)
  81| 
  82|     # Instance-folder file fallback (survives restarts on the same instance)
  83|     try:
  84|         key_file = Path(app.instance_path) / ".secret_key"
  85|         if key_file.exists():
  86|             stored = key_file.read_text().strip()
  87|             if stored:
  88|                 app.logger.info("SECRET_KEY auto-provisioned (file — reuse)")
  89|                 return stored
  90|         new_key = secrets.token_hex(32)
  91|         key_file.write_text(new_key)
  92|         app.logger.info("SECRET_KEY auto-provisioned (file — new)")
  93|         return new_key
  94|     except OSError as exc:
  95|         app.logger.warning("SECRET_KEY file fallback unavailable: %s", exc)
  96| 
  97|     # Ephemeral last resort — never block a deploy; sessions reset on restart
  98|     app.logger.warning(
  99|         "SECRET_KEY not set and no persistence available — using ephemeral key; "
 100|         "sessions will reset on restart. Provision SECRET_KEY in Render "
 101|         "(render.yaml generateValue: true for new services, or set it in the "
 102|         "dashboard for existing services).",
 103|     )
 104|     return secrets.token_hex(32)

 353|     # Fallback safeguard: if core tables are missing (e.g., fresh local DB
 354|     # without migrations applied), create them so startup sync doesn't fail.
 355|     # On a FRESH database we also stamp the Alembic head: the historical
 356|     # migration chain was written as incremental patches on top of a
 357|     # db.create_all()-created schema (e.g. the baseline adds columns to
 358|     # tables that don't exist yet from migrations alone), so replaying it
 359|     # against a fresh DB crashes on duplicate columns. Stamping makes the
 360|     # subsequent `flask db upgrade` in the Render start command a no-op
 361|     # while future migrations still apply normally.
 362|     with app.app_context():
 363|         from sqlalchemy import create_engine
 364|         from sqlalchemy import inspect as sa_inspect
 365| 
 366|         engine = create_engine(app.config["SQLALCHEMY_DATABASE_URI"])
 367|         inspector = sa_inspect(engine)
 368|         if "fso" not in inspector.get_table_names():
 369|             db.create_all()
 370|             app.logger.info("Created missing tables via db.create_all() fallback")
 371|             # Only stamp when there is NO migration history at all — never
 372|             # clobber a partially-migrated database.
 373|             if "alembic_version" not in inspector.get_table_names():
 374|                 try:
 375|                     from flask_migrate import stamp as alembic_stamp
 376| 
 377|                     alembic_stamp(revision="head")
 378|                     app.logger.info("Stamped fresh database at migration head")
 379|                 except Exception as exc:
 380|                     app.logger.warning(
 381|                         "Could not stamp fresh database at migration head (%s) — "
 382|                         "`flask db upgrade` may replay the full chain next deploy.",
 383|                         exc,
 384|                     )
